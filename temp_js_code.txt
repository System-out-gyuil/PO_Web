// 기대출 금액 업데이트 함수
function updateLoanAmount(rowId, institution, amount) {
    // 현재 행의 모든 기대출 데이터 수집
    const loanData = {};
    const loanInstitutions = ['기술보증기금', '신용보증기금', '신용보증재단', '중진공', '소진공', '기타1', '기타2', '기타3'];
    
    loanInstitutions.forEach(function(inst) {
        const input = document.getElementById(`loan-${rowId}-${inst}`);
        if (input) {
            loanData[inst] = input.value || '';
        }
    });
    
    // 서버에 전체 기대출 데이터 전송
    fetch('/diary/update_loan_amount/', {
        method: 'POST',
        headers: {
            'Content-Type': 'application/x-www-form-urlencoded',
            'X-CSRFToken': document.querySelector('[name=csrfmiddlewaretoken]').value
        },
        body: `row_id=${encodeURIComponent(rowId)}&loan_data=${encodeURIComponent(JSON.stringify(loanData))}`
    })
    .then(response => response.json())
    .then(data => {
        if (data.success) {
            console.log('기대출 금액이 성공적으로 저장되었습니다.');
            // 테이블 새로고침
            refreshTable();
        } else {
            console.error('기대출 금액 저장 실패:', data.error);
            alert('기대출 금액 저장에 실패했습니다: ' + (data.error || '알 수 없는 오류'));
        }
    })
    .catch(error => {
        console.error('기대출 금액 저장 오류:', error);
        alert('기대출 금액 저장 중 오류가 발생했습니다.');
    });
}
